cmake_minimum_required(VERSION 3.22)
project(duga_vulkan LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/build/shaders/shaders.hpp")
  message(FATAL_ERROR "Compile shaders first: node native/vulkan/shaders.mjs (requires pinned naga-cli 30.0.1)")
endif()
add_library(duga_vulkan SHARED vulkan.cpp gguf.cpp dense.cpp jni.cpp oracle.cpp)
target_include_directories(duga_vulkan PRIVATE build/shaders)
target_compile_definitions(duga_vulkan PRIVATE DUGA_NO_ORACLE_MAIN)
target_compile_options(duga_vulkan PRIVATE -ffp-contract=off -Wall -Wextra -Wno-missing-field-initializers)
target_link_options(duga_vulkan PRIVATE "-Wl,-z,max-page-size=16384")
target_link_libraries(duga_vulkan vulkan log)
